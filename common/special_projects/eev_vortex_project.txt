# Constructor Rescue
special_project = {
	key = "EEV_SCI_REPAIR"
	cost = 0
	days_to_research = 50
	tech_department = engineering_technology
	timelimit = 1080
	picture = GFX_evt_ship_in_orbit

	event_scope = ship_event

	requirements = {
		shipclass_constructor = 1
	}
	
	on_success = {
		# Ship scope is the constructor completing the rescue project.
		ship_event = { id = eev_houston.2 }
	}

	on_cancel = {
		event_target:eev_vortex_sci = {
			ship_event = { id = eev_houston.6 days = 50 }
		}
	}

	on_fail = {
		event_target:eev_vortex_sci = {
			ship_event = { id = eev_houston.6 days = 50 }
		}
	}
}

special_project = {
	key = "EEV_MIL_REPAIR2"
	cost = 0
	days_to_research = 50
	tech_department = engineering_technology
	timelimit = 1080
	picture = GFX_evt_ship_in_orbit

	event_scope = ship_event

	requirements = {
		shipclass_military = 1
	}
	
	on_success = {
		# Record the military rescuer, not the stranded science ship.
		ship_event = { id = eev_houston.4 }
	}
	
	on_cancel = {
		event_target:eev_vortex_con = {
			if = {
				limit = { is_ship_class = shipclass_constructor }
				fleet = { destroy_fleet = this }
			}
			else = { fleet = { set_event_locked = no } }
		}
		event_target:eev_vortex_sci = {
			ship_event = { id = eev_houston.6 days = 50 }
		}
	}

	on_fail = {
		event_target:eev_vortex_con = {
			if = {
				limit = { is_ship_class = shipclass_constructor }
				fleet = { destroy_fleet = this }
			}
			else = { fleet = { set_event_locked = no } }
		}
		event_target:eev_vortex_sci = {
			ship_event = { id = eev_houston.6 days = 50 }
		}
	}
}
